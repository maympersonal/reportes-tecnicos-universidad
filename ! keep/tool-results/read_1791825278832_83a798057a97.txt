    35→function fmtFecha(s?: string | null) {
    36→  if (!s) return "—"
    37→  try {
    38→    return new Date(s).toLocaleString("es-ES", {
    39→      day: "2-digit",
    40→      month: "short",
    41→      year: "numeric",
    42→      hour: "2-digit",
    43→      minute: "2-digit",
    44→    })
    45→  } catch {
    46→    return "—"
    47→  }
    48→}
